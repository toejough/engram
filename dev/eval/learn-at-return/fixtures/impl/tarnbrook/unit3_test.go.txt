package lock

import (
	"errors"
	"os"
	"path/filepath"
	"strconv"
	"strings"
	"testing"
)

func TestAcquire(t *testing.T) {
	t.Parallel()

	t.Run("creates lock with pid", func(t *testing.T) {
		t.Parallel()
		dir := t.TempDir()
		held, err := Acquire(dir)
		if err != nil {
			t.Fatal(err)
		}
		defer held.Release()
		body, err := os.ReadFile(filepath.Join(dir, Name))
		if err != nil || strings.TrimSpace(string(body)) != strconv.Itoa(os.Getpid()) {
			t.Fatalf("lock body %q, err %v", body, err)
		}
	})

	t.Run("held lock names pid", func(t *testing.T) {
		t.Parallel()
		dir := t.TempDir()
		if err := os.WriteFile(filepath.Join(dir, Name), []byte("4242\n"), 0o644); err != nil {
			t.Fatal(err)
		}
		_, err := Acquire(dir)
		if !errors.Is(err, ErrHeld) || !strings.Contains(err.Error(), "pid 4242") {
			t.Fatalf("want ErrHeld naming pid 4242, got %v", err)
		}
	})
}

func TestRelease(t *testing.T) {
	t.Parallel()

	t.Run("removes lock", func(t *testing.T) {
		t.Parallel()
		dir := t.TempDir()
		held, err := Acquire(dir)
		if err != nil {
			t.Fatal(err)
		}
		if err := held.Release(); err != nil {
			t.Fatal(err)
		}
		if _, err := os.Stat(filepath.Join(dir, Name)); !errors.Is(err, os.ErrNotExist) {
			t.Fatalf("lock still present: %v", err)
		}
	})

	t.Run("missing lock is ok", func(t *testing.T) {
		t.Parallel()
		held := &Lock{path: filepath.Join(t.TempDir(), Name)}
		if err := held.Release(); err != nil {
			t.Fatal(err)
		}
	})
}
